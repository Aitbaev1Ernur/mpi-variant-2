#include <mpi.h>
#include <errno.h>
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static const char *input;

static void require(int ok) {
    if (!ok) {
        int rank;
        MPI_Comm_rank(MPI_COMM_WORLD, &rank);
        fprintf(stderr, "%d-ранг. Кіріс деректері немесе процесс саны жарамсыз\n", rank);
        MPI_Abort(MPI_COMM_WORLD, 1);
        exit(EXIT_FAILURE);
    }
}

static int integer(void) {
    char *end;
    errno = 0;
    long value = strtol(input, &end, 10);
    require(end != input && errno == 0 && value >= INT_MIN && value <= INT_MAX);
    input = end;
    return (int)value;
}

static double real(void) {
    char *end;
    errno = 0;
    double value = strtod(input, &end);
    require(end != input && errno == 0);
    input = end;
    return value;
}

static void vd(int rank, const double *a, int n) {
    size_t size = 32 * ((size_t)n + 1);
    char *line = malloc(size);
    require(line != NULL);
    size_t used = (size_t)snprintf(line, size, "%d:", rank);
    for (int i = 0; i < n; i++)
        used += (size_t)snprintf(line + used, size - used, " %.12g", a[i]);
    puts(line);
    free(line);
}

int main(int argc, char **argv) {
    int r, k;
    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &r);
    MPI_Comm_size(MPI_COMM_WORLD, &k);
    require(argc == k + 1);
    input = argv[r + 1];

    if (r % 2 == 0) {
        int a = integer();
        printf("%d: %lld\n", r, 2LL * a);
    } else {
        double a = real();
        a *= 2;
        vd(r, &a, 1);
    }

    require(input[strspn(input, " \t\r\n")] == '\0');
    MPI_Finalize();
    return 0;
}

    int index[k], edges[8 * n], used = 0;
    for (int i = 0; i < k; i++) {
        if (i == 0) {
            for (int j = 3; j < k; j += 3) edges[used++] = j;

            int first = 1 + 3 * ((i - 1) / 3);
