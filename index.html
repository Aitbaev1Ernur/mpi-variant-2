Тапсырмаларды қалай іске қосуға болады?

Тапсырманы ашқанда мысалдың нәтижесі бірден көрсетіледі. Өз сандарыңызды тексеру үшін кіріс деректерін өзгертіп, «Іске қосу» түймесін басыңыз.

Әр жол 0-рангтен бастап бір процеске сәйкес келеді. Сандарды бос орынмен бөліңіз. Кіріс деректері жоқ процесс үшін — белгісін қойыңыз. Нәтижеде қос нүктенің алдында ранг көрсетіледі.

Браузер MPI процестерінің дерек алмасуын модельдейді. C тіліндегі бағдарламалар «Бастапқы код» бөлімінде берілген.

Теория

Екі процесс арасында хабарлама жіберу

8 тапсырма

Теория

Екі процесс арасында хабарлама жіберу үшін MPI_Send және MPI_Recv қолданылады. Жіберуші деректің түрін, санын, алушы рангін және тегін көрсетеді. Қабылдаушы сол коммуникатордағы үйлесімді хабарламаны дерек түрі, жіберуші және тег бойынша қабылдайды. Қабылдау буфері хабарламаға жеткілікті болуы керек.

MPI_ANY_SOURCE кез келген жіберушіден, MPI_ANY_TAG кез келген тегпен қабылдауға мүмкіндік береді. MPI_Probe хабарламаны алмай тұрып оның мәртебесін анықтайды, ал MPI_Get_count сол мәртебе бойынша элементтер санын береді. MPI_Sendrecv_replace жіберу мен қабылдауды бір шақыруда орындап, бастапқы буферді қабылданған мәндермен толтырады.

Редукция операциялары және құрама дерек түрлері

9 тапсырма

Теория

Редукция бірнеше процестің дерегін бір операциямен біріктіреді. Ұзындықтары бірдей тізімдерде операция әр позицияға жеке қолданылады. MPI_Reduce нәтижені бір процеске, MPI_Allreduce барлық қатысушыға береді. MPI_Reduce_scatter нәтижені бөліп таратады, ал MPI_Scan әр ранг үшін 0-ден сол рангке дейінгі деректерді біріктіреді.

Туынды MPI дерек түрі бірнеше өрісті бір жазба ретінде жіберуге мүмкіндік береді. Мысалы, бір жазбада екі бүтін және бір нақты сан болуы мүмкін. Дерек түрін қолданар алдында MPI_Type_commit арқылы бекітеміз. MPI_Pack бірнеше мәнді бір буферге жинайды, ал MPI_Unpack оларды жазылған ретпен шығарады.

Мысал. Екі процесте [2, 7] және [4, 3] тізімдері берілсе, MPI_MIN нәтижесі [2, 3], ал MPI_PROD нәтижесі [8, 21] болады.

Ұжымдық операциялар және коммуникаторлар құру

11 тапсырма

Теория

Ұжымдық операцияны берілген коммуникатордың барлық қатысушысы бірдей ретпен шақырады. MPI_Gather деректерді бір процеске жинайды, MPI_Allgather жиналған деректерді барлық қатысушыға береді. MPI_Scatterv деректерді әр рангке бөлек берілген көлем және ығысу бойынша таратады. MPI_Alltoall және MPI_Alltoallv кезінде әр процесс әр қатысушыға дерек жібере алады.

Жаңа коммуникатор процестердің бір бөлігіне өзара жұмыс істеуге мүмкіндік береді. MPI_Group_excl кейбір рангтерді топтан алып тастайды, ал MPI_Comm_split процестерді белгі бойынша бөледі. Жаңа коммуникатордағы жергілікті ранг MPI_COMM_WORLD ішіндегі рангтен өзгеше болуы мүмкін.

Мысал. 1 және 3 рангтері ғана таңдалса, олар жаңа коммуникаторда тиісінше 0 және 1 рангтерін ала алады. Сонда MPI_Gather операциясының жергілікті 0-рангі бастапқы 1-рангке сәйкес келеді.

Виртуалды топология процестерді тор немесе граф түрінде ұйымдастырады. MPI_Cart_create декарттық торды құрады. MPI_Cart_coords ранг бойынша координатаны, MPI_Cart_rank координата бойынша рангті анықтайды. Өлшем периодты болса, тордың соңғы ұяшығынан кейін қайтадан бірінші ұяшық келеді.

#include <mpi.h>
#include <errno.h>
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static const char *input;

static void require(int ok) {
    if (!ok) {
        int rank;
        MPI_Comm_rank(MPI_COMM_WORLD, &rank);
        fprintf(stderr, "%d-ранг. Кіріс деректері немесе процесс саны жарамсыз\n", rank);
        MPI_Abort(MPI_COMM_WORLD, 1);
        exit(EXIT_FAILURE);
    }
}

static int integer(void) {
    char *end;
    errno = 0;
    long value = strtol(input, &end, 10);
    require(end != input && errno == 0 && value >= INT_MIN && value <= INT_MAX);
    input = end;
    return (int)value;
}

static double real(void) {
    char *end;
    errno = 0;
    double value = strtod(input, &end);
    require(end != input && errno == 0);
    input = end;
    return value;
}

static void vd(int rank, const double *a, int n) {
    size_t size = 32 * ((size_t)n + 1);
    char *line = malloc(size);
    require(line != NULL);
    size_t used = (size_t)snprintf(line, size, "%d:", rank);
    for (int i = 0; i < n; i++)
        used += (size_t)snprintf(line + used, size - used, " %.12g", a[i]);
    puts(line);
    free(line);
}

int main(int argc, char **argv) {
    int r, k;
    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &r);
    MPI_Comm_size(MPI_COMM_WORLD, &k);
    require(argc == k + 1);
    input = argv[r + 1];

    if (r % 2 == 0) {
        int a = integer();
        printf("%d: %lld\n", r, 2LL * a);
    } else {
        double a = real();
        a *= 2;
        vd(r, &a, 1);
    }

    require(input[strspn(input, " \t\r\n")] == '\0');
    MPI_Finalize();
    return 0;
}

        MPI_Abort(MPI_COMM_WORLD, 1);
        exit(EXIT_FAILURE);
    }
}

static int integer(void) {
    char *end;
    errno = 0;
    long value = strtol(input, &end, 10);
    require(end != input && errno == 0 && value >= INT_MIN && value <= INT_MAX);
    input = end;
    return (int)value;
}

static void vi(int rank, const int *a, int n) {
    size_t size = 32 * ((size_t)n + 1);
    char *line = malloc(size);
    require(line != NULL);
    size_t used = (size_t)snprintf(line, size, "%d:", rank);
    for (int i = 0; i < n; i++)
        used += (size_t)snprintf(line + used, size - used, " %d", a[i]);
    puts(line);
    free(line);
}

int main(int argc, char **argv) {
    int r, k;
    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &r);
    MPI_Comm_size(MPI_COMM_WORLD, &k);
    require(argc == k + 1);
    input = argv[r + 1];

    int n = (k - 1) / 3;
    require(k == 3 * n + 1 && n >= 2 && n <= 4);
    int index[k], edges[8 * n], used = 0;
    for (int i = 0; i < k; i++) {
        if (i == 0) {
            for (int j = 3; j < k; j += 3) edges[used++] = j;
        } else {
            int first = 1 + 3 * ((i - 1) / 3);
            if (i % 3 == 0) edges[used++] = 0;
            for (int j = first; j < first + 3; j++)
                if (j != i) edges[used++] = j;
        }
        index[i] = used;
    }
    MPI_Comm graph;
    MPI_Graph_create(MPI_COMM_WORLD, k, index, edges, 0, &graph);
    int count, a = integer();
    MPI_Graph_neighbors_count(graph, r, &count);
    int neighbors[count], out[count];
    MPI_Graph_neighbors(graph, r, count, neighbors);
    for (int i = 1; i < count; i++) {
        int j = i, peer = neighbors[i];
        while (j > 0 && neighbors[j - 1] > peer) {
            neighbors[j] = neighbors[j - 1];
            j--;
        }
        neighbors[j] = peer;
    }
    for (int i = 0; i < count; i++) {
        int peer = neighbors[i];
        MPI_Sendrecv(&a, 1, MPI_INT, peer, 0, out + i, 1, MPI_INT, peer, 0,
                     graph, MPI_STATUS_IGNORE);
    }
    vi(r, out, count);
    MPI_Comm_free(&graph);

    require(input[strspn(input, " \t\r\n")] == '\0');
    MPI_Finalize();
    return 0;
}
